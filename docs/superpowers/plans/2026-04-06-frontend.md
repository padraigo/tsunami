```ts
import type {
  Simulation,
  FocusZone,
  CoarseResult,
  Preset,
  CreateSimulationPayload,
  CreateFocusZonePayload,
} from '../types'

// ---------------------------------------------------------------------------
// Base fetch helper — throws on non-2xx, returns parsed JSON
// ---------------------------------------------------------------------------
async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`/api${path}`, {
    headers: { 'Content-Type': 'application/json', ...init?.headers },
    ...init,
  })
  if (!res.ok) {
    const text = await res.text().catch(() => res.statusText)
    throw new Error(`API ${res.status}: ${text}`)
  }
  // DELETE returns 204 No Content
  if (res.status === 204) return undefined as unknown as T
  return res.json() as Promise<T>
}

// ---------------------------------------------------------------------------
// Simulations
// ---------------------------------------------------------------------------
export const api = {
  simulations: {
    list: () => apiFetch<Simulation[]>('/simulations'),
    create: (payload: CreateSimulationPayload) =>
      apiFetch<Simulation>('/simulations', {
        method: 'POST',
        body: JSON.stringify(payload),
      }),
    get: (uid: string) => apiFetch<Simulation>(`/simulations/${uid}`),
    delete: (uid: string) =>
      apiFetch<void>(`/simulations/${uid}`, { method: 'DELETE' }),
  },

  // Focus zones
  focusZones: {
    list: (uid: string) =>
      apiFetch<FocusZone[]>(`/simulations/${uid}/focus-zones`),
    create: (uid: string, payload: CreateFocusZonePayload) =>
      apiFetch<FocusZone>(`/simulations/${uid}/focus-zones`, {
        method: 'POST',
        body: JSON.stringify(payload),
      }),
    update: (simUid: string, zoneUid: string, payload: Partial<CreateFocusZonePayload>) =>
      apiFetch<FocusZone>(`/simulations/${simUid}/focus-zones/${zoneUid}`, {
        method: 'PUT',
        body: JSON.stringify(payload),
      }),
    delete: (simUid: string, zoneUid: string) =>
      apiFetch<void>(`/simulations/${simUid}/focus-zones/${zoneUid}`, {
        method: 'DELETE',
      }),
  },

  // Run + results
  run: {
    coarse: (uid: string) =>
      apiFetch<CoarseResult>(`/simulations/${uid}/run-coarse`, { method: 'POST' }),
    coarseResult: (uid: string) =>
      apiFetch<CoarseResult>(`/simulations/${uid}/coarse-result`),
    exportGeoJSON: (uid: string) => `/api/simulations/${uid}/export?format=geojson`,
  },

  // Bathymetry
  bathymetry: {
    check: (bounds: { lat_min: number; lat_max: number; lon_min: number; lon_max: number }) => {
      const q = new URLSearchParams({
        lat_min: String(bounds.lat_min),
        lat_max: String(bounds.lat_max),
        lon_min: String(bounds.lon_min),
        lon_max: String(bounds.lon_max),
      })
      return apiFetch<{ available: boolean }>(`/bathymetry/check?${q}`)
    },
  },

  // Presets
  presets: {
    locations: () => apiFetch<Preset[]>('/presets/locations'),
  },
}
```

```ts
import type { WsMessage } from '../types'

export type WsListener = (msg: WsMessage) => void

export class SimulationWebSocket {
  private ws: WebSocket | null = null
  private listeners = new Set<WsListener>()
  private uid: string | null = null

  connect(uid: string): void {
    if (this.ws) this.disconnect()
    this.uid = uid
    const proto = location.protocol === 'https:' ? 'wss' : 'ws'
    const url = `${proto}://${location.host}/api/ws/simulations/${uid}`
    this.ws = new WebSocket(url)

    this.ws.onmessage = (event) => {
      try {
        const msg = JSON.parse(event.data) as WsMessage
        this.listeners.forEach((fn) => fn(msg))
      } catch {
        console.warn('[WS] Failed to parse message', event.data)
      }
    }

    this.ws.onerror = (err) => console.error('[WS] Error', err)
    this.ws.onclose = () => {
      if (this.uid === uid) this.ws = null
    }
  }

  disconnect(): void {
    this.ws?.close()
    this.ws = null
    this.uid = null
  }

  subscribe(fn: WsListener): () => void {
    this.listeners.add(fn)
    return () => this.listeners.delete(fn)
  }

  get connected(): boolean {
    return this.ws?.readyState === WebSocket.OPEN
  }
}

// Singleton — one connection at a time
export const simulationWS = new SimulationWebSocket()
```

```ts
import { create } from 'zustand'
import { api } from '../services/api'
import { simulationWS } from '../services/websocket'
import type {
  Simulation,
  CoarseResult,
  Preset,
  WsMessage,
  CreateSimulationPayload,
  CreateFocusZonePayload,
  SuggestedZone,
} from '../types'

interface SimulationState {
  // Data
  simulations: Simulation[]
  selectedUid: string | null
  coarseResult: CoarseResult | null
  presets: Preset[]
  error: string | null
  loading: boolean

  // Derived
  selected: Simulation | undefined

  // Actions
  loadSimulations: () => Promise<void>
  selectSimulation: (uid: string) => Promise<void>
  createSimulation: (payload: CreateSimulationPayload) => Promise<Simulation>
  deleteSimulation: (uid: string) => Promise<void>
  runCoarse: (uid: string) => Promise<void>
  loadCoarseResult: (uid: string) => Promise<void>
  addFocusZone: (uid: string, payload: CreateFocusZonePayload) => Promise<void>
  addSuggestedZone: (uid: string, zone: SuggestedZone) => Promise<void>
  deleteFocusZone: (simUid: string, zoneUid: string) => Promise<void>
  loadPresets: () => Promise<void>
  handleWsMessage: (msg: WsMessage) => void
  clearError: () => void
}

export const useSimulationStore = create<SimulationState>((set, get) => ({
  simulations: [],
  selectedUid: null,
  coarseResult: null,
  presets: [],
  error: null,
  loading: false,

  get selected() {
    return get().simulations.find((s) => s.uid === get().selectedUid)
  },

  loadSimulations: async () => {
    set({ loading: true, error: null })
    try {
      const simulations = await api.simulations.list()
      set({ simulations, loading: false })
    } catch (e) {
      set({ error: String(e), loading: false })
    }
  },

  selectSimulation: async (uid: string) => {
    set({ selectedUid: uid, coarseResult: null, loading: true, error: null })
    try {
      const sim = await api.simulations.get(uid)
      set((state) => ({
        simulations: state.simulations.map((s) => (s.uid === uid ? sim : s)),
        loading: false,
      }))
      simulationWS.connect(uid)
      simulationWS.subscribe(get().handleWsMessage)
      // Load existing result if available
      try {
        const result = await api.run.coarseResult(uid)
        set({ coarseResult: result })
      } catch {
        // no result yet — that's fine
      }
    } catch (e) {
      set({ error: String(e), loading: false })
    }
  },

  createSimulation: async (payload) => {
    set({ loading: true, error: null })
    try {
      const sim = await api.simulations.create(payload)
      set((state) => ({ simulations: [sim, ...state.simulations], loading: false }))
      return sim
    } catch (e) {
      set({ error: String(e), loading: false })
      throw e
    }
  },

  deleteSimulation: async (uid) => {
    await api.simulations.delete(uid)
    set((state) => ({
      simulations: state.simulations.filter((s) => s.uid !== uid),
      selectedUid: state.selectedUid === uid ? null : state.selectedUid,
      coarseResult: state.selectedUid === uid ? null : state.coarseResult,
    }))
  },

  runCoarse: async (uid) => {
    set({ loading: true, error: null })
    try {
      // Update optimistic status
      set((state) => ({
        simulations: state.simulations.map((s) =>
          s.uid === uid ? { ...s, status: 'running_coarse' } : s
        ),
      }))
      const result = await api.run.coarse(uid)
      set((state) => ({
        simulations: state.simulations.map((s) =>
          s.uid === uid ? { ...s, status: result.status } : s
        ),
        coarseResult: result,
        loading: false,
      }))
    } catch (e) {
      set({ error: String(e), loading: false })
    }
  },

  loadCoarseResult: async (uid) => {
    try {
      const result = await api.run.coarseResult(uid)
      set({ coarseResult: result })
    } catch (e) {
      set({ error: String(e) })
    }
  },

  addFocusZone: async (uid, payload) => {
    const zone = await api.focusZones.create(uid, payload)
    set((state) => ({
      simulations: state.simulations.map((s) =>
        s.uid === uid ? { ...s, focus_zones: [...s.focus_zones, zone] } : s
      ),
    }))
  },

  addSuggestedZone: async (uid, zone) => {
    const payload: CreateFocusZonePayload = {
      name: `Auto zone (${zone.max_impact_height.toFixed(1)} m)`,
      lat_min: zone.lat_min,
      lat_max: zone.lat_max,
      lon_min: zone.lon_min,
      lon_max: zone.lon_max,
    }
    await get().addFocusZone(uid, payload)
  },

  deleteFocusZone: async (simUid, zoneUid) => {
    await api.focusZones.delete(simUid, zoneUid)
    set((state) => ({
      simulations: state.simulations.map((s) =>
        s.uid === simUid
          ? { ...s, focus_zones: s.focus_zones.filter((z) => z.uid !== zoneUid) }
          : s
      ),
    }))
  },

  loadPresets: async () => {
    try {
      const presets = await api.presets.locations()
      set({ presets })
    } catch (e) {
      set({ error: String(e) })
    }
  },

  handleWsMessage: (msg: WsMessage) => {
    if (msg.type === 'status_update' && msg.status) {
      set((state) => ({
        simulations: state.simulations.map((s) =>
          s.uid === msg.uid ? { ...s, status: msg.status! } : s
        ),
      }))
    }
    if (msg.type === 'result_ready') {
      get().loadCoarseResult(msg.uid)
    }
  },

  clearError: () => set({ error: null }),
}))
```

```tsx
import { useEffect, useRef, useCallback } from 'react'
import maplibregl from 'maplibre-gl'
import { useSimulationStore } from '../../stores/simulationStore'
import type { ImpactPoint, FocusZone } from '../../types'

// Free raster tile source — no API key required
const BASEMAP_STYLE: maplibregl.StyleSpecification = {
  version: 8,
  sources: {
    'osm-tiles': {
      type: 'raster',
      tiles: ['https://tile.openstreetmap.org/{z}/{x}/{y}.png'],
      tileSize: 256,
      attribution: '© OpenStreetMap contributors',
      maxzoom: 19,
    },
  },
  layers: [{ id: 'osm-tiles', type: 'raster', source: 'osm-tiles' }],
}

interface MapViewProps {
  onEpicenterClick?: (lat: number, lon: number) => void
}

export function MapView({ onEpicenterClick }: MapViewProps) {
  const mapContainer = useRef<HTMLDivElement>(null)
  const map = useRef<maplibregl.Map | null>(null)
  const epicenterMarker = useRef<maplibregl.Marker | null>(null)

  const { selected, coarseResult } = useSimulationStore()

  // Initialise map once
  useEffect(() => {
    if (!mapContainer.current || map.current) return

    map.current = new maplibregl.Map({
      container: mapContainer.current,
      style: BASEMAP_STYLE,
      center: [160, 5],    // Centre on Pacific
      zoom: 2,
    })

    map.current.addControl(new maplibregl.NavigationControl(), 'top-right')

    // Click to set epicentre
    map.current.on('click', (e) => {
      onEpicenterClick?.(e.lngLat.lat, e.lngLat.lng)
    })

    return () => {
      map.current?.remove()
      map.current = null
    }
  }, [])

  // Update epicenter marker when selected sim changes
  useEffect(() => {
    if (!map.current || !selected) return

    epicenterMarker.current?.remove()
    const el = document.createElement('div')
    el.className = 'w-4 h-4 rounded-full bg-red-500 border-2 border-white shadow-lg cursor-move'

    epicenterMarker.current = new maplibregl.Marker({ element: el, draggable: true })
      .setLngLat([selected.earthquake_lon, selected.earthquake_lat])
      .addTo(map.current)

    // Pan to epicenter
    map.current.flyTo({
      center: [selected.earthquake_lon, selected.earthquake_lat],
      zoom: 5,
      duration: 1200,
    })
  }, [selected?.uid, selected?.earthquake_lat, selected?.earthquake_lon])

  // Draw impact points
  const drawImpacts = useCallback((impacts: ImpactPoint[]) => {
    if (!map.current) return
    const m = map.current

    const sourceId = 'impact-points'
    const layerId = 'impact-circles'

    if (m.getLayer(layerId)) m.removeLayer(layerId)
    if (m.getSource(sourceId)) m.removeSource(sourceId)

    if (impacts.length === 0) return

    const maxH = Math.max(...impacts.map((p) => p.max_height), 1)

    m.addSource(sourceId, {
      type: 'geojson',
      data: {
        type: 'FeatureCollection',
        features: impacts.map((pt) => ({
          type: 'Feature',
          geometry: { type: 'Point', coordinates: [pt.lon, pt.lat] },
          properties: {
            max_height: pt.max_height,
            arrival_time_s: pt.arrival_time_s,
            // Normalised 0–1 for colour ramp
            intensity: pt.max_height / maxH,
          },
        })),
      },
    })

    m.addLayer({
      id: layerId,
      type: 'circle',
      source: sourceId,
      paint: {
        'circle-radius': ['interpolate', ['linear'], ['get', 'intensity'], 0, 4, 1, 12],
        'circle-color': [
          'interpolate',
          ['linear'],
          ['get', 'intensity'],
          0,   '#ffffb2',
          0.25,'#fecc5c',
          0.5, '#fd8d3c',
          0.75,'#f03b20',
          1,   '#bd0026',
        ],
        'circle-opacity': 0.85,
        'circle-stroke-width': 1,
        'circle-stroke-color': '#ffffff',
      },
    })
  }, [])

  // Draw focus zone rectangles
  const drawZones = useCallback((zones: FocusZone[]) => {
    if (!map.current) return
    const m = map.current

    const sourceId = 'focus-zones'
    const fillId = 'focus-zones-fill'
    const lineId = 'focus-zones-line'
    const labelId = 'focus-zones-label'

    for (const id of [fillId, lineId, labelId]) if (m.getLayer(id)) m.removeLayer(id)
    if (m.getSource(sourceId)) m.removeSource(sourceId)

    if (zones.length === 0) return

    m.addSource(sourceId, {
      type: 'geojson',
      data: {
        type: 'FeatureCollection',
        features: zones.map((z) => ({
          type: 'Feature',
          geometry: {
            type: 'Polygon',
            coordinates: [[
              [z.lon_min, z.lat_min],
              [z.lon_max, z.lat_min],
              [z.lon_max, z.lat_max],
              [z.lon_min, z.lat_max],
              [z.lon_min, z.lat_min],
            ]],
          },
          properties: { name: z.name, status: z.status },
        })),
      },
    })

    m.addLayer({ id: fillId, type: 'fill', source: sourceId, paint: { 'fill-color': '#3b82f6', 'fill-opacity': 0.15 } })
    m.addLayer({ id: lineId, type: 'line', source: sourceId, paint: { 'line-color': '#3b82f6', 'line-width': 2 } })
    m.addLayer({
      id: labelId,
      type: 'symbol',
      source: sourceId,
      layout: { 'text-field': ['get', 'name'], 'text-size': 11, 'text-anchor': 'center' },
      paint: { 'text-color': '#93c5fd', 'text-halo-color': '#1e293b', 'text-halo-width': 1 },
    })
  }, [])

  // Re-draw when results or zones change
  useEffect(() => {
    if (!map.current) return
    // Wait until the map style is loaded
    if (!map.current.isStyleLoaded()) {
```

```tsx
function formatArrival(seconds: number): string {
  const h = Math.floor(seconds / 3600)
  const m = Math.floor((seconds % 3600) / 60)
  return h > 0 ? `${h}h ${m}m` : `${m}m`
}

export function ImpactList({ impacts }: ImpactListProps) {
  if (impacts.length === 0) {
    return <p className="text-slate-500">No impact data yet. Run a coarse simulation first.</p>
  }

  // Sort by max_height descending
  const sorted = [...impacts].sort((a, b) => b.max_height - a.max_height)

  return (
    <table className="w-full text-xs">
      <thead>
        <tr className="border-b border-slate-700 text-left text-slate-400">
          <th className="pb-1 pr-3">Lat</th>
          <th className="pb-1 pr-3">Lon</th>
          <th className="pb-1 pr-3">Height (m)</th>
          <th className="pb-1">Arrival</th>
        </tr>
      </thead>
      <tbody>
        {sorted.slice(0, 50).map((pt, i) => (
          <tr
            key={i}
            className="border-b border-slate-800 hover:bg-slate-700/40"
          >
            <td className="py-0.5 pr-3 font-mono">{pt.lat.toFixed(3)}</td>
```
